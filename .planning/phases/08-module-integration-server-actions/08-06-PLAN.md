    ```typescript
    @injectable()
    export class DrizzleAuthQueryRepository implements IQueryAuthRepository {
      constructor(
        @inject(AUTH_TOKENS.DRIZZLE_CLIENT) private readonly db: DrizzleClient,
      ) {}

      async findUserByEmail(email: string): Promise<DomainResult<UserAggregate | null>> {
        try {
          const result = await withRetry(
            () => withTimeout(
              () => this.db.select().from(usersTable).where(eq(usersTable.email, email)).limit(1),
              10_000,
              { systemComponent: "Database", errorCode: "DB_TIMEOUT" },
            ),
            3,
            { systemComponent: "Database", errorCode: "DB_RETRY_EXHAUSTED" },
          );

          return result.match<DomainResult<UserAggregate | null>>({
            onSuccess: (rows) => {
              const row = rows[0];
              return row ? ok(mapUserRowToAggregate(row)) : ok(null);
            },
            onFailure: (error) => err(error as any), // Convert to DomainResult compatible
          });
        } catch (error) {
          return err(new DatabaseQueryError({
            message: `Failed to find user by email: ${(error as Error).message}`,
            cause: error as Error,
            query: "findUserByEmail",
          }) as any);
        }
      }

      async findUserById(id: string): Promise<DomainResult<UserAggregate | null>> {
        try {
          const result = await withRetry(
            () => withTimeout(
              () => this.db.select().from(usersTable).where(eq(usersTable.id, id)).limit(1),
              10_000,
              { systemComponent: "Database", errorCode: "DB_TIMEOUT" },
            ),
            3,
            { systemComponent: "Database", errorCode: "DB_RETRY_EXHAUSTED" },
          );

          return result.match<DomainResult<UserAggregate | null>>({
            onSuccess: (rows) => {
              const row = rows[0];
              return row ? ok(mapUserRowToAggregate(row)) : ok(null);
            },
            onFailure: (error) => err(error as any),
          });
        } catch (error) {
          return err(new DatabaseQueryError({
            message: `Failed to find user by id: ${(error as Error).message}`,
            cause: error as Error,
            query: "findUserById",
          }) as any);
        }
      }
    }
    ```

    ```typescript
    @injectable()
    export class DrizzleAuthCommandRepository implements ICommandAuthRepository {
      constructor(
        @inject(AUTH_TOKENS.DRIZZLE_CLIENT) private readonly db: DrizzleClient,
      ) {}

      async createUser(data: UserAggregateCreateInput): Promise<DomainResult<UserAggregate>> {
        try {
          const result = await withRetry(
            () => withTimeout(
              () => this.db.insert(usersTable).values({ ...data }).returning(),
              10_000,
              { systemComponent: "Database", errorCode: "DB_TIMEOUT" },
            ),
            3,
            { systemComponent: "Database", errorCode: "DB_RETRY_EXHAUSTED" },
          );

          return result.match<DomainResult<UserAggregate>>({
            onSuccess: (rows) => {
              const row = rows[0];
              if (!row) {
                return err(new DatabaseQueryError({
                  message: "User creation returned no rows",
                  query: "createUser",
                }) as any);
              }
              return ok(mapUserRowToAggregate(row));
            },
            onFailure: (error) => {
              // Check for unique constraint violation (PostgreSQL error code 23505)
              const pgError = error as any;
              if (pgError?.code === "23505" || pgError?.message?.includes("duplicate key")) {
                return err(new DatabaseQueryError({
                  message: "Email already exists",
                  query: "createUser",
                  cause: pgError,
                }) as any);
              }
              return err(error as any);
            },
          });
        } catch (error) {
          return err(new DatabaseQueryError({
            message: `Failed to create user: ${(error as Error).message}`,
            cause: error as Error,
            query: "createUser",
          }) as any);
        }
      }
    }
    ```
