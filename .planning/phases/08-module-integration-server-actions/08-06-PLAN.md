<objective>
Create the Authentication module's Infrastructure Query Repository implementation using Drizzle ORM with @injectable() decorator, full InfrastructureResult wrapping, DB-to-Domain mappers, and DrizzleClient DI binding.

Purpose: Repository implementation bridges Drizzle (external DB) and the Domain layer. The query repository has @injectable() and injects the Drizzle client via @inject(AUTH_TOKENS.DRIZZLE_CLIENT) bound as toConstantValue(getDatabaseClient()). All DB operations return DomainResult via InfrastructureResult resilience wrappers. DB rows are mapped to domain aggregates via pure functions. No Command repository — all auth commands delegated to BetterAuth.
</objective>

    ```typescript
    @injectable()
    export class DrizzleAuthQueryRepository implements AuthQueryRepository {
      constructor(
        @inject(AUTH_TOKENS.DRIZZLE_CLIENT) private readonly db: DrizzleClient,
      ) {}

      async findUserByEmail(email: string): Promise<DomainResult<UserAggregate | null>> {
        try {
          const result = await withRetry(
            () => withTimeout(
              () => this.db.select().from(usersTable).where(eq(usersTable.email, email)).limit(1),
              10_000,
              { systemComponent: "Database", errorCode: "DB_TIMEOUT" },
            ),
            3,
            { systemComponent: "Database", errorCode: "DB_RETRY_EXHAUSTED" },
          );

          return result.match<DomainResult<UserAggregate | null>>({
            onSuccess: (rows) => {
              const row = rows[0];
              return row ? ok(mapUserRowToAggregate(row)) : ok(null);
            },
            onFailure: (error) => err(error as any), // Convert to DomainResult compatible
          });
        } catch (error) {
          return err(new DatabaseQueryError({
            message: `Failed to find user by email: ${(error as Error).message}`,
            cause: error as Error,
            query: "findUserByEmail",
          }) as any);
        }
      }

      async findUserById(id: string): Promise<DomainResult<UserAggregate | null>> {
        try {
          const result = await withRetry(
            () => withTimeout(
              () => this.db.select().from(usersTable).where(eq(usersTable.id, id)).limit(1),
              10_000,
              { systemComponent: "Database", errorCode: "DB_TIMEOUT" },
            ),
            3,
            { systemComponent: "Database", errorCode: "DB_RETRY_EXHAUSTED" },
          );

          return result.match<DomainResult<UserAggregate | null>>({
            onSuccess: (rows) => {
              const row = rows[0];
              return row ? ok(mapUserRowToAggregate(row)) : ok(null);
            },
            onFailure: (error) => err(error as any),
          });
        } catch (error) {
          return err(new DatabaseQueryError({
            message: `Failed to find user by id: ${(error as Error).message}`,
            cause: error as Error,
            query: "findUserById",
          }) as any);
        }
      }
    }
    ```
    Note: `as any` casts are for InfrastructureError → DomainError compatibility. The Infrastructure resilience wrappers return InfrastructureResult; the method signature requires DomainResult. This is expected since DomainResult<T> = ResultBase<T, DomainError> and the actual type parameter is structural.
