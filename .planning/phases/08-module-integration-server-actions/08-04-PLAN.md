```typescript
// src/Modules/Authentication/Domain/Errors/error-codes.constants.ts
export const AUTH_ERROR_CODES = {
  USER_NOT_FOUND: "AUTH_USER_NOT_FOUND",
  INVALID_CREDENTIALS: "AUTH_INVALID_CREDENTIALS",
  EMAIL_ALREADY_EXISTS: "AUTH_EMAIL_ALREADY_EXISTS",
} as const;
```

**Domain error pattern (Foundation positional params + constants):**
```typescript
import { DomainError } from "@/Core/Foundations/Domain";
import { AUTH_ERROR_CODES } from "./error-codes.constants";
// No InversifyJS imports needed — errors are not resolved via DI

// Foundation DomainError signature:
// constructor(code: string, message: string, businessRule: string, aggregateId: string, cause?: DomainErrorContract)

export class UserNotFoundError extends DomainError {
  constructor(message: string, businessRule: string, aggregateId: string, cause?: DomainError) {
    super(AUTH_ERROR_CODES.USER_NOT_FOUND, message, businessRule, aggregateId, cause);
  }
}
```

**Repository Interface pattern (pure TypeScript — no decorators):**
```typescript
// Interfaces are pure TypeScript contracts — NOT decorated with @injectable()
// InversifyJS binds Symbol.for('Auth.IQueryAuthRepository') → DrizzleAuthQueryRepository

import type { DomainResult } from "@/Core/Foundations/Domain/Results";
import type { UserAggregate } from "../../Aggregates/user.aggregate";

export interface IQueryAuthRepository {
  findUserByEmail(email: string): Promise<DomainResult<UserAggregate | null>>;
  findUserById(id: string): Promise<DomainResult<UserAggregate | null>>;
}
```

**InversifyJS binding pattern (done in 08-06 in registration — noop here):**
