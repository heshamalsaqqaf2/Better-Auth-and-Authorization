**Error code constants pattern (avoid hardcoded strings):**
```typescript
// src/Modules/Authentication/Domain/Errors/error-codes.constants.ts
export const AUTH_ERROR_CODES = {
  USER_NOT_FOUND: "AUTH_USER_NOT_FOUND",
  INVALID_CREDENTIALS: "AUTH_INVALID_CREDENTIALS",
  EMAIL_ALREADY_EXISTS: "AUTH_EMAIL_ALREADY_EXISTS",
  AUTH_FAILED: "AUTH_FAILED",
} as const;

export type AuthErrorCode = (typeof AUTH_ERROR_CODES)[keyof typeof AUTH_ERROR_CODES];
```

**Domain error pattern (Foundation positional params + constants):**
```typescript
import { DomainError } from "@/Core/Foundations/Domain";
import { AUTH_ERROR_CODES } from "./error-codes.constants";

export class UserNotFoundError extends DomainError {
  constructor(message: string, businessRule: string, aggregateId: string, cause?: DomainError) {
    super(AUTH_ERROR_CODES.USER_NOT_FOUND, message, businessRule, aggregateId, cause);
  }
}
```

**Aggregate types pattern (pure interfaces — typed data contracts):**
```typescript
// src/Modules/Authentication/Domain/Aggregates/authenticated-user.ts
export interface AuthenticatedUser {
  id: string;
  name: string;
  email: string;
  emailVerified: boolean;
  image?: string | null;
  createdAt: Date;
  updatedAt: Date;
}

export interface AuthSession {
  id: string;
  userId: string;
  expiresAt: Date;
  createdAt: Date;
  ipAddress?: string | null;
  userAgent?: string | null;
}
// Note: token is excluded from AuthSession — managed by BetterAuth, not exposed via Domain
```

**Repository Interface pattern (pure TypeScript — no decorators, no I prefix):**
```typescript
// Interfaces are pure TypeScript contracts — NOT decorated with @injectable()
// InversifyJS binds Symbol.for('Auth.AuthQueryRepository') → DrizzleAuthQueryRepository

import type { DomainResult } from "@/Core/Foundations/Domain/Results";
import type { AuthenticatedUser } from "../../Aggregates/authenticated-user";

export interface AuthQueryRepository {
  findByEmail(email: string): Promise<DomainResult<AuthenticatedUser | null>>;
  findById(id: string): Promise<DomainResult<AuthenticatedUser | null>>;
}
```
