**UseCase injection pattern (InversifyJS with explicit @inject):**
```typescript
import { injectable, inject } from "inversify";
import { AUTH_TOKENS } from "@/Modules/Authentication/Composition/tokens";

@injectable()
export class SignInUseCase {
  constructor(
    @inject(AUTH_TOKENS.BETTER_AUTH_SERVICE) private readonly authService: IBetterAuthService,
    @inject(AUTH_TOKENS.QUERY_AUTH_REPOSITORY) private readonly userRepo: IQueryAuthRepository,
  ) {}

  async execute(dto: SignInDTO): Promise<ApplicationResult<AuthResponseDTO>> {
    // 1. authService.signIn(email, password) → InfrastructureResult
    // 2. If Failure: err(mapInfraToAppError(failure))
    // 3. If Success: userRepo.findUserByEmail(email) → DomainResult
    // 4. If Failure: err(mapDomainToAppError(failure))
    // 5. Map to AuthResponseDTO → ok(dto)
    // 6. try/catch → err(UseCaseExecutionError)
  }
}
```

**Why explicit @inject() is required:**
- Next.js 16 uses SWC for transpilation
- SWC supports `experimentalDecorators` (legacy decorator transform) but NOT `emitDecoratorMetadata`
- Without metadata emission, InversifyJS cannot auto-resolve constructor parameter types
- Each constructor parameter MUST have an explicit `@inject(TOKEN)` decorator

**DTO shapes:**
```typescript
interface SignInDTO { email: string; password: string; }
interface SignUpDTO { name: string; email: string; password: string; }
interface AuthResponseDTO { user: UserDTO; session: SessionDTO; }
interface UserDTO { id: string; name: string; email: string; }  // No sensitive fields
interface SessionDTO { id: string; expiresAt: string; }  // ISO string, not Date
interface SignOutDTO {}  // Empty marker — side-effect only
interface SessionCheckDTO { user: UserDTO | null; session: SessionDTO | null; }
```

      ```typescript
      @injectable()
      export class SignUpUseCase {
        constructor(
          @inject(AUTH_TOKENS.BETTER_AUTH_SERVICE) private readonly authService: IBetterAuthService,
          @inject(AUTH_TOKENS.COMMAND_AUTH_REPOSITORY) private readonly cmdRepo: ICommandAuthRepository,
          @inject(AUTH_TOKENS.QUERY_AUTH_REPOSITORY) private readonly queryRepo: IQueryAuthRepository,
        ) {}

        async execute(dto: SignUpDTO): Promise<ApplicationResult<AuthResponseDTO>> {
          try {
            // 1. Check for duplicate email
            const existing = await this.queryRepo.findUserByEmail(dto.email);
            if (existing.isSuccess() && existing.data !== null) {
              return err(mapDomainToAppError(new EmailAlreadyExistsError({ email: dto.email })));
            }

            // 2. Sign up via Better Auth
            const signUpResult = await this.authService.signUp(dto.name, dto.email, dto.password);
            if (signUpResult.isFailure()) {
              return err(mapInfraToAppError(signUpResult.error));
            }

            // 3. Store user in local repository
            const userResult = await this.cmdRepo.createUser({
              name: dto.name,
              email: dto.email,
            });
            if (userResult.isFailure()) {
              return err(mapDomainToAppError(userResult.error));
            }

            // 4. Map to response
            return ok(mapAuthResultToResponse(userResult.data, signUpResult.data as any));
          } catch (error) {
            return err(new UseCaseExecutionError({ cause: error instanceof Error ? error : new Error(String(error)), operationName: "signUp", correlationId: "" }));
          }
        }
      }
      ```
